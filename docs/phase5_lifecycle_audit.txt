from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Any
from uuid import UUID, uuid4


class TaskEventType(str, Enum):
    TASK_CREATED = "task_created"
    STATUS_CHANGED = "status_changed"
    TASK_FAILED = "task_failed"


@dataclass(frozen=True)
class TaskEvent:
    id: UUID
    task_id: UUID
    event_type: TaskEventType
    timestamp: datetime
    message: str
    metadata: dict[str, Any] = field(default_factory=dict)

    @classmethod
    def create(
        cls,
        task_id: UUID,
        event_type: TaskEventType,
        message: str,
        metadata: dict[str, Any] | None = None,
    ) -> "TaskEvent":
        cleaned_message = message.strip()

        if not cleaned_message:
            raise ValueError("Event message cannot be empty.")

        return cls(
            id=uuid4(),
            task_id=task_id,
            event_type=event_type,
            timestamp=datetime.now(),
            message=cleaned_message,
            metadata=dict(metadata) if metadata is not None else {},
        )

from datetime import datetime
from uuid import UUID

from orion.core.tasks.event_repository import TaskEventRepository
from orion.core.tasks.events import TaskEvent, TaskEventType
from orion.core.tasks.models import Task, TaskStatus
from orion.core.tasks.repository import TaskRepository

ALLOWED_TRANSITIONS: dict[TaskStatus, set[TaskStatus]] = {
    TaskStatus.CREATED: {
        TaskStatus.PLANNING,
        TaskStatus.CANCELLED,
    },
    TaskStatus.PLANNING: {
        TaskStatus.RUNNING,
        TaskStatus.CANCELLED,
    },
    TaskStatus.RUNNING: {
        TaskStatus.WAITING,
        TaskStatus.VERIFYING,
        TaskStatus.FAILED,
        TaskStatus.CANCELLED,
    },
    TaskStatus.WAITING: {
        TaskStatus.RUNNING,
        TaskStatus.CANCELLED,
    },
    TaskStatus.VERIFYING: {
        TaskStatus.COMPLETED,
        TaskStatus.FAILED,
    },
    TaskStatus.FAILED: {
        TaskStatus.PLANNING,
        TaskStatus.CANCELLED,
    },
    TaskStatus.COMPLETED: set(),
    TaskStatus.CANCELLED: set(),
}


class TaskManager:
    def __init__(
        self,
        repository: TaskRepository,
        event_repository: TaskEventRepository,
    ) -> None:
        self._repository = repository
        self._event_repository = event_repository

    def create_task(self, objective: str) -> Task:
        task = Task.create(objective)
        self._repository.save(task)

        event = TaskEvent.create(
            task_id=task.id,
            event_type=TaskEventType.TASK_CREATED,
            message="Task created",
        )
        self._event_repository.save(event)

        return task

    def get_task(self, task_id: UUID) -> Task | None:
        return self._repository.get(task_id)

    def update_status(
        self,
        task_id: UUID,
        status: TaskStatus,
        error: str | None = None,
    ) -> Task:
        task = self._repository.get(task_id)

        if task is None:
            raise ValueError(f"Task not found: {task_id}")

        if status != TaskStatus.FAILED and error is not None:
            raise ValueError(
                "An error message can only be provided when "
                "transitioning a task to FAILED."
            )

        allowed_statuses = ALLOWED_TRANSITIONS[task.status]

        if status not in allowed_statuses:
            raise ValueError(
                f"Invalid task transition: "
                f"{task.status.value} -> {status.value}"
            )

        previous_status = task.status

        if status == TaskStatus.FAILED:
            cleaned_error = error.strip() if error is not None else None

            if not cleaned_error:
                raise ValueError(
                    "An error message is required when a task fails."
                )

            task.last_error = cleaned_error

        elif status == TaskStatus.PLANNING:
            task.last_error = None

        now = datetime.now()

        if status == TaskStatus.RUNNING:
            task.started_at = task.started_at or now
            task.attempt_count += 1

        if status == TaskStatus.COMPLETED:
            task.completed_at = now

        task.status = status
        self._repository.save(task)

        if status == TaskStatus.FAILED:
            event_type = TaskEventType.TASK_FAILED
            message = f"Task failed: {task.last_error}"
        else:
            event_type = TaskEventType.STATUS_CHANGED
            message = (
                f"Task status changed: "
                f"{previous_status.value} -> {status.value}"
            )

        event = TaskEvent.create(
            task_id=task.id,
            event_type=event_type,
            message=message,
            metadata={
                "previous_status": previous_status.value,
                "new_status": status.value,
                "attempt_count": task.attempt_count,
            },
        )
        self._event_repository.save(event)

        return task

from orion.config.model_config import ModelConfig
from orion.core.agents.execution_service import AgentExecutionService
from orion.core.agents.repository import AgentRepository
from orion.core.agents.tasks import AgentTask
from orion.core.recovery.failure_classifier import FailureClassifier
from orion.models.contracts import ModelRequest
from orion.models.runtime.runtime import ModelRuntime


class AgentExecutor:
    """Executes an assigned ORION agent task using the model runtime."""

    def __init__(
        self,
        agent_repository: AgentRepository,
        execution_service: AgentExecutionService,
        model_runtime: ModelRuntime,
        failure_classifier: FailureClassifier | None = None,
    ) -> None:
        self._agent_repository = agent_repository
        self._execution_service = execution_service
        self._model_runtime = model_runtime
        self._failure_classifier = (
            failure_classifier or FailureClassifier()
        )

    def execute(
        self,
        task: AgentTask,
        model_config: ModelConfig,
    ):
        """Execute an assigned agent task."""

        if task.assigned_agent_id is None:
            raise ValueError(
                f"Agent task '{task.id}' has no assigned agent."
            )

        agent = self._agent_repository.get(task.assigned_agent_id)

        if agent is None:
            raise ValueError(
                f"Agent '{task.assigned_agent_id}' was not found."
            )

        execution = self._execution_service.create_execution(
            agent_task_id=task.id,
            agent_id=agent.id,
        )

        request = ModelRequest(
            prompt=task.description,
            metadata={
                "agent_id": str(agent.id),
                "agent_name": agent.name,
                "agent_task_id": str(task.id),
            },
        )

        try:
            response = self._model_runtime.generate(
                model_config,
                request,
            )

            return self._execution_service.complete_execution(
                execution_id=execution.id,
                output=response.text,
                model_name=response.model_name,
                provider_name=response.provider_name,
                input_tokens=response.input_tokens,
                output_tokens=response.output_tokens,
                latency_ms=response.latency_ms,
            )

        except Exception as exc:
            failure = self._failure_classifier.classify(exc)

            return self._execution_service.fail_execution(
                execution_id=execution.id,
                failure=failure,
            )

class AgentAssigner:
    """Assigns agent tasks to capable ORION agents."""

    def __init__(
        self,
        capability_registry: AgentCapabilityRegistry,
    ) -> None:
        self._capability_registry = capability_registry

    def assign(self, task: AgentTask) -> AgentAssignment:
        """Assign a task to an available agent with the required capability."""

        agents = self._capability_registry.find_by_capability(
            task.required_capability
        )

        if not agents:
            raise ValueError(
                f"No agent available with capability "
                f"'{task.required_capability.value}'."
            )

        agent = agents[0]

        return AgentAssignment.create(
            agent=agent,
            capability=task.required_capability,
        )

        return updated_task

from orion.core.agents.capabilities import AgentCapability
from orion.core.agents.models import Agent


class AgentCapabilityRegistry:
    """Find agents based on their capabilities."""

    def __init__(self) -> None:
        self._agents: list[Agent] = []

    def register(self, agent: Agent) -> None:
        """Register an agent for capability-based lookup."""

        if agent in self._agents:
            raise ValueError(
                f"Agent '{agent.name}' is already registered."
            )

        self._agents.append(agent)

    def find_by_capability(
        self,
        capability: AgentCapability,
    ) -> list[Agent]:
        """Return all agents that have the requested capability."""

        return [
            agent
            for agent in self._agents
            if capability in agent.capabilities
        ]

    def list_capabilities(self) -> frozenset[AgentCapability]:
        """Return every capability provided by registered agents."""

        capabilities: set[AgentCapability] = set()

        for agent in self._agents:
            capabilities.update(agent.capabilities)

        return frozenset(capabilities)

from dataclasses import replace
from datetime import datetime
from uuid import UUID

from orion.core.agents.execution_manager import AgentExecutionManager
from orion.core.agents.execution_repository import AgentExecutionRepository
from orion.core.agents.executions import (
    AgentExecution,
    AgentExecutionStatus,
)
from orion.core.recovery.failures import Failure, FailureType


class AgentExecutionService:
    """Coordinates creation and lifecycle updates of agent executions."""

    def __init__(
        self,
        repository: AgentExecutionRepository,
    ) -> None:
        self._repository = repository
        self._manager = AgentExecutionManager()

    def create_execution(
        self,
        agent_task_id: UUID,
        agent_id: UUID,
    ) -> AgentExecution:
        """Create and persist a new running execution."""

        execution = AgentExecution.create(
            agent_task_id=agent_task_id,
            agent_id=agent_id,
        )

        self._repository.save(execution)

        return execution

    def get_execution(
        self,
        execution_id: UUID,
    ) -> AgentExecution | None:
        """Get an execution by ID."""

        return self._repository.get(execution_id)

    def update_status(
        self,
        execution_id: UUID,
        status: AgentExecutionStatus,
    ) -> AgentExecution:
        """Update and persist an execution status."""

        execution = self._repository.get(execution_id)

        if execution is None:
            raise ValueError(
                f"Agent execution '{execution_id}' not found."
            )

        updated_execution = self._manager.update_status(
            execution,
            status,
        )

        self._repository.save(updated_execution)

        return updated_execution

    def complete_execution(
        self,
        execution_id: UUID,
        output: str,
        model_name: str,
        provider_name: str,
        input_tokens: int | None = None,
        output_tokens: int | None = None,
        latency_ms: float | None = None,
    ) -> AgentExecution:
        """Complete an execution and persist its result metadata."""

        execution = self._repository.get(execution_id)

        if execution is None:
            raise ValueError(
                f"Agent execution '{execution_id}' not found."
            )

        updated_execution = self._manager.update_status(
            execution,
            AgentExecutionStatus.COMPLETED,
        )

        updated_execution = replace(
            updated_execution,
            completed_at=datetime.now(),
            output=output,
            model_name=model_name,
            provider_name=provider_name,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            latency_ms=latency_ms,
        )

        self._repository.save(updated_execution)

        return updated_execution

    def fail_execution(
        self,
        execution_id: UUID,
        error_message: str | None = None,
        failure: Failure | None = None,
    ) -> AgentExecution:
        """Fail an execution and persist its error and structured failure."""

        if failure is not None and not isinstance(failure, Failure):
            raise TypeError(
                "failure must be a Failure instance."
            )

        if failure is None:
            if error_message is None or not error_message.strip():
                raise ValueError(
                    "Either error_message or failure must be provided."
                )

            failure = Failure(
                failure_type=FailureType.UNKNOWN,
                message=error_message,
            )

        execution = self._repository.get(execution_id)

        if execution is None:
            raise ValueError(
                f"Agent execution '{execution_id}' not found."
            )

        updated_execution = self._manager.update_status(
            execution,
            AgentExecutionStatus.FAILED,
        )

        updated_execution = replace(
            updated_execution,
            completed_at=datetime.now(),
            error_message=failure.message,
            failure=failure,
        )

        self._repository.save(updated_execution)

        return updated_execution

    def list_for_agent_task(
        self,
        agent_task_id: UUID,
    ) -> list[AgentExecution]:
        """List all execution attempts for an agent task."""

        return self._repository.list_for_agent_task(agent_task_id)

@dataclass(frozen=True)
class ExecutionLoopResult:
    """Result of executing a collection of assigned agent tasks."""

    executions: tuple[AgentExecution, ...]


class ExecutionLoop:
    """Coordinates task execution and bounded iterative recovery."""

    DEFAULT_MAX_RECOVERY_ATTEMPTS = 3

    def __init__(
        self,
        execution_coordinator: ExecutionCoordinator,
        parallel_executor: ParallelExecutor | None = None,
        recovery_coordinator: RecoveryCoordinator | None = None,
        recovery_service: RecoveryService | None = None,
        state_synchronizer: StateSynchronizer | None = None,
        max_recovery_attempts: int = DEFAULT_MAX_RECOVERY_ATTEMPTS,
    ) -> None:
        if not isinstance(max_recovery_attempts, int):
            raise TypeError("max_recovery_attempts must be an integer.")

        if max_recovery_attempts < 0:
            raise ValueError(
                "max_recovery_attempts cannot be negative."
            )

        self._execution_coordinator = execution_coordinator
        self._parallel_executor = parallel_executor
        self._recovery_coordinator = recovery_coordinator
        self._recovery_service = recovery_service
        self._state_synchronizer = state_synchronizer
        self._max_recovery_attempts = max_recovery_attempts

    def execute(
        self,
        tasks: list[AgentTask],
        model_config: ModelConfig,
        plan_steps: list[PlanStep] | None = None,
        alternative_model_configs: tuple[ModelConfig, ...] = (),
    ) -> ExecutionLoopResult:
        """Execute tasks and continue through bounded recovery when possible."""

        executions: list[AgentExecution] = []

        if self._parallel_executor is not None:
            result = self._parallel_executor.execute(
                tasks=tasks,
                model_config=model_config,
            )

            for execution in result.executions:
                task = self._find_task(
                    tasks,
                    execution.agent_task_id,
                )

                executions.extend(
                    self._process_execution_with_recovery(
                        execution=execution,
                        task=task,
                        model_config=model_config,
                        alternative_model_configs=alternative_model_configs,
                        plan_steps=plan_steps,
                    )
                )

            return ExecutionLoopResult(
                executions=tuple(executions),
            )

        for task in tasks:
            execution = self._execution_coordinator.execute_task(
                task=task,
                model_config=model_config,
            )

            executions.extend(
                self._process_execution_with_recovery(
                    execution=execution,
                    task=task,
                    model_config=model_config,
                    alternative_model_configs=alternative_model_configs,
                    plan_steps=plan_steps,
                )
            )

        return ExecutionLoopResult(
            executions=tuple(executions),
        )

    def _process_execution_with_recovery(
        self,
        execution: AgentExecution,
        task: AgentTask,
        model_config: ModelConfig,
        alternative_model_configs: tuple[ModelConfig, ...],
        plan_steps: list[PlanStep] | None,
    ) -> list[AgentExecution]:
        """Process an execution and iteratively apply bounded recovery."""

        executions = [execution]

        self._synchronize_execution(
            execution=execution,
            plan_steps=plan_steps,
        )

        if execution.status != AgentExecutionStatus.FAILED:
            return executions

        if self._recovery_coordinator is None:
            return executions

        if execution.failure is None:
            return executions

        context = RecoveryExecutionContext.from_execution(
            execution
        )

        current_execution = execution
        current_model_config = model_config

        while True:
            if current_execution.status != AgentExecutionStatus.FAILED:
                break

            if current_execution.failure is None:
                break

            if (
                context.recovery_attempt_number
                >= self._max_recovery_attempts
            ):
                break

            recovery_result = self._recovery_coordinator.handle_failure(
                execution=current_execution,
                execution_context=context,
                max_retries=self._max_recovery_attempts,
                alternative_models_available=bool(
                    alternative_model_configs
                ),
            )

            context = context.record_recovery(
                recovery_result.decision
            )

            action_type = recovery_result.action.action_type

            if recovery_result.action.requires_human:
                break

            if self._recovery_service is None:
                break

            if action_type not in {
                RecoveryActionType.RETRY_TASK,
                RecoveryActionType.CHANGE_MODEL,
            }:
                break

            self._synchronize_failed_task_before_retry(
                execution=current_execution,
                task=task,
                plan_steps=plan_steps,
            )

            if action_type == RecoveryActionType.RETRY_TASK:
                next_execution = self._recovery_service.retry_task(
                    task,
                    current_model_config,
                )

            else:
                next_execution = (
                    self._recovery_service.change_model_and_retry(
                        task=task,
                        current_model_config=current_model_config,
                        alternative_model_configs=alternative_model_configs,
                    )
                )

                if next_execution is not None:
                    current_model_config = (
                        self._model_config_after_change(
                            current_model_config=current_model_config,
                            alternative_model_configs=alternative_model_configs,
                        )
                    )

            if next_execution is None:
                break

            executions.append(next_execution)

            context = context.create_next_attempt(
                current_execution_id=next_execution.id,
            )

            current_execution = next_execution

            self._synchronize_execution(
                execution=current_execution,
                plan_steps=plan_steps,
            )

            if current_execution.status != AgentExecutionStatus.FAILED:
                break

            if current_execution.failure is None:
                break

            context = context.record_failure(
                current_execution.failure
            )

        return executions

    @staticmethod
    def _model_config_after_change(
        current_model_config: ModelConfig,
        alternative_model_configs: tuple[ModelConfig, ...],
    ) -> ModelConfig:
        """Return the selected alternative model configuration."""

        for alternative in alternative_model_configs:
            if (
                alternative.provider_name
                != current_model_config.provider_name
            ):
                return alternative

        return current_model_config

    def _synchronize_failed_task_before_retry(
        self,
        execution: AgentExecution,
        task: AgentTask,
        plan_steps: list[PlanStep] | None,
    ) -> None:
        """Synchronize failed state before retrying when synchronization is configured."""

        if task.status == AgentTaskStatus.FAILED:
            return

        if self._state_synchronizer is None:
            return

        if plan_steps is None:
            raise ValueError(
                "plan_steps are required before retrying a failed task."
            )

        self._state_synchronizer.synchronize(
            execution,
            plan_steps,
        )

    def _synchronize_execution(
        self,
        execution: AgentExecution,
        plan_steps: list[PlanStep] | None,
    ) -> None:
        """Synchronize execution state when a synchronizer is configured."""

        if self._state_synchronizer is None:
            return

        if plan_steps is None:
            raise ValueError(
                "plan_steps are required when state_synchronizer is configured."
            )

        self._state_synchronizer.synchronize(
            execution,
            plan_steps,
        )

    @staticmethod
    def _find_task(
        tasks: list[AgentTask],
        task_id,
    ) -> AgentTask:
        """Find the task associated with an execution."""

        for task in tasks:
            if task.id == task_id:
                return task

        raise ValueError(
            f"Agent task not found: {task_id}"
        )

class Orchestrator:
    """Coordinates plan steps, agent tasks, and execution."""

    def __init__(
        self,
        scheduler: PlanScheduler,
        agent_task_service: AgentTaskService,
        assignment_service: AgentAssignmentService,
        execution_loop: ExecutionLoop | None = None,
    ) -> None:
        self._scheduler = scheduler
        self._agent_task_service = agent_task_service
        self._assignment_service = assignment_service
        self._execution_loop = execution_loop

        task_id: UUID,
        plan_steps: list[PlanStep],

        ready_steps = self._scheduler.get_ready_steps(plan_steps)

        agent_tasks: list[AgentTask] = []

        for step in ready_steps:
            agent_task = self._agent_task_service.create_task(
                task_id=task_id,
                plan_step_id=step.id,
                description=step.description,
                required_capability=step.required_capability,

                agent_task
            )

            agent_tasks.append(assigned_task)

        return OrchestrationResult(
            ready_steps=tuple(ready_steps),
            agent_tasks=tuple(agent_tasks),
        )

        self,
        task_id: UUID,
        plan_steps: list[PlanStep],
        model_config: ModelConfig,
    ) -> OrchestrationExecutionResult:
        """Orchestrate ready tasks and execute them."""

        if self._execution_loop is None:
            raise ValueError(
                "Execution loop is required for orchestrate_and_execute."
            )

        orchestration = self.orchestrate(
            task_id=task_id,
            plan_steps=plan_steps,
        )

        execution = self._execution_loop.execute(
            tasks=list(orchestration.agent_tasks),
            model_config=model_config,
        )

        return OrchestrationExecutionResult(
            orchestration=orchestration,
            execution=execution,
        )

from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass

from orion.config.model_config import ModelConfig
from orion.core.agents.executions import AgentExecution
from orion.core.agents.tasks import AgentTask
from orion.core.orchestration.execution_coordinator import ExecutionCoordinator


@dataclass(frozen=True)
class ParallelExecutionResult:
    """Result of executing agent tasks in parallel."""

    executions: tuple[AgentExecution, ...]


class ParallelExecutor:
    """Executes independent assigned agent tasks with bounded concurrency."""

    DEFAULT_MAX_WORKERS = 4

    def __init__(
        self,
        execution_coordinator: ExecutionCoordinator,
        max_workers: int = DEFAULT_MAX_WORKERS,
    ) -> None:
        if max_workers <= 0:
            raise ValueError("max_workers must be greater than zero.")

        self._execution_coordinator = execution_coordinator
        self._max_workers = max_workers

    @property
    def max_workers(self) -> int:
        """Return the maximum number of concurrent workers."""
        return self._max_workers

    def execute(
        self,
        tasks: list[AgentTask],
        model_config: ModelConfig,
    ) -> ParallelExecutionResult:
        """Execute tasks concurrently while preserving input result order."""

        if not tasks:
            return ParallelExecutionResult(executions=())

        with ThreadPoolExecutor(max_workers=self._max_workers) as executor:
            futures = [
                executor.submit(
                    self._execution_coordinator.execute_task,
                    task,
                    model_config,
                )
                for task in tasks
            ]

            executions = tuple(
                future.result()
                for future in futures
            )

        return ParallelExecutionResult(
            executions=executions,
        )

    ) -> None:
        self._task_service = task_service

    def synchronize(
        self,
        execution: AgentExecution,
        plan_steps: list[PlanStep],
    ) -> None:
        """Synchronize an execution result with its agent task and plan step."""

        task = self._task_service.get_task(execution.agent_task_id)

        if task is None:
            raise ValueError(
                f"Agent task not found: {execution.agent_task_id}"
            )

        plan_step = next(
            (
                step
                for step in plan_steps
                if step.id == task.plan_step_id
            ),
            None,
        )

        if plan_step is None:
            raise ValueError(
                f"Plan step not found: {task.plan_step_id}"
            )

        if execution.status == AgentExecutionStatus.RUNNING:
            self._task_service.update_status(
                task.id,
                AgentTaskStatus.RUNNING,
            )
            plan_step.status = PlanStepStatus.RUNNING

        elif execution.status == AgentExecutionStatus.COMPLETED:
            self._task_service.update_status(
                task.id,
                AgentTaskStatus.COMPLETED,
            )
            plan_step.status = PlanStepStatus.COMPLETED

        elif execution.status == AgentExecutionStatus.FAILED:
            self._task_service.update_status(
                task.id,
                AgentTaskStatus.FAILED,
            )
            plan_step.status = PlanStepStatus.FAILED

        elif execution.status == AgentExecutionStatus.CANCELLED:
            self._task_service.update_status(
                task.id,
                AgentTaskStatus.CANCELLED,
            )
            plan_step.status = PlanStepStatus.SKIPPED
